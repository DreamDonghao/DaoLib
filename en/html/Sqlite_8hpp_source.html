154 /// @tparam Args 参数类型
155 /// @param sql SQL 查询语句，使用 {} 作为占位符
156 /// @param args 参数值
157 /// @return 查询结果集
158 template<typename... Args>
159 Result select(std::string_view sql, Args &&... args) {
160 Result result;
161 if (!m_database) {
162 Log{LogLevel::DEBUG}("数据库未打开");
163 return result;
164 }
165 try {
166 const std::string formattedSql = formatSql(sql, std::forward<Args>(args)...);
167 SQLite::Statement query(*m_database, formattedSql);
168 // 获取列名
169 std::vector<std::string> columnNames;
170 const int columnCount = query.getColumnCount();
171 columnNames.reserve(columnCount);
172 for (int i = 0; i < columnCount; ++i) {
173 columnNames.push_back(query.getColumnName(i));
174 }
175
176 // 遍历结果
177 while (query.executeStep()) {
178 std::map<std::string, DataVal> rowData;
179 for (int i = 0; i < columnCount; ++i) {
180 if (const auto col = query.getColumn(i); col.isNull()) {
181 rowData[columnNames[i]] = DataVal("");
182 } else {
183 rowData[columnNames[i]] = DataVal(col.getText());
184 }
185 }
186 result.addRow(Row(std::move(rowData)));
187 }
188 } catch (const SQLite::Exception &e) {
189 DAO_ERROR_LOG(std::string("查询失败: ") + e.what());
190 }
191 return result;
192 }

241 template<typename... Args>
242 static std::string formatSql(const std::string_view sql, Args &&... args) {
243 std::string result(sql);
244 size_t pos = 0;
245 ((pos = replaceNext(result, pos, std::forward<Args>(args))), ...);

249 template<typename T>
250 static size_t replaceNext(std::string &sql, const size_t pos, T &&value) {
251 const size_t found = sql.find("{}", pos);
252 if (found == std::string::npos) {
253 return std::string::npos;
254 }
255 const auto escaped = escapeValue(std::forward<T>(value));
256 sql.replace(found, 2, escaped);
257 return found + escaped.size();
258 }
