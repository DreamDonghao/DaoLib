41 void handleEvent(const SDL_Event &event) {
42 if (m_status == InputStatus::Disabled) {
43 return;
44 }
45 float x = 0, y = 0;
46 if (event.type == SDL_EVENT_MOUSE_MOTION) {
47 x = event.motion.x;
48 y = event.motion.y;
49 } else if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN || event.type == SDL_EVENT_MOUSE_BUTTON_UP) {
50 x = event.button.x;
51 y = event.button.y;
52 }
53
54 SDL_Window *win = SDL_GetWindowFromID(event.window.windowID);
55 if (m_status == InputStatus::Input) {
56 const SDL_Rect rect{
57 static_cast<int>(m_boundingBox.getLeft()),
58 static_cast<int>(m_boundingBox.getTop()), // 顶部
59 static_cast<int>(m_boundingBox.getWidth()), // 宽度
60 static_cast<int>(m_boundingBox.getHeight()) // 高度
61 };
62 SDL_SetTextInputArea(win, &rect, cursorPos);
63 if (event.type == SDL_EVENT_TEXT_INPUT) {
64 const char *utf8chars = event.text.text;
65 const auto begin = utf8::unchecked::iterator(utf8chars);
66 const auto end = utf8::unchecked::iterator(utf8chars + std::strlen(utf8chars));
67
68 for (auto it = begin; it != end; ++it) {
69 m_str.insert(m_str.begin() + cursorPos, static_cast<char32_t>(*it));
70 cursorPos += 1;
71 }
72 } else if (event.type == SDL_EVENT_KEY_DOWN) {
73 if (event.key.key == SDLK_BACKSPACE && !m_str.empty()) {
74 if (cursorPos != 0) {
75 m_str.erase(cursorPos - 1, 1);
76 --cursorPos;
77 }
78 } else if (event.key.key == SDLK_DELETE) {
79 if (cursorPos != m_str.size()) {
80 m_str.erase(cursorPos, 1);
81 }
82 } else if (event.key.key == SDLK_LEFT) {
83 cursorPos = std::max(cursorPos - 1, 0);
84 } else if (event.key.key == SDLK_RIGHT) {
85 cursorPos = std::min(cursorPos + 1, static_cast<int>(m_str.size()));
86 }
87 } else if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN && event.button.button == SDL_BUTTON_LEFT) {
88 if (!m_boundingBox.isInBoundingBox(x, y)) {
89 m_status = InputStatus::Normal;
90 SDL_StopTextInput(win);
91 }
92 }
93 } else if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN && event.button.button == SDL_BUTTON_LEFT) {
94 if (m_boundingBox.isInBoundingBox(x, y)) {
95 m_status = InputStatus::Input;
96 SDL_StartTextInput(win);
97 }
98 }
99 }
