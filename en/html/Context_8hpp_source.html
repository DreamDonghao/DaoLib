20 /// @tparam T 对象类型

22 /// @param args 若为单个 T 类型参数，则引用该对象；否则用于构造新对象
23 /// @returns 对象的引用

26 template<typename T, typename... Args>
27 T &emplace(Args &&... args) {
28 if constexpr (sizeof...(args) == 1 &&
