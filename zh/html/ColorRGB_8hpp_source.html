116 /// @brief 将十六进制颜色字符串转换为ColorRGBA对象
117 /// @param hex 十六进制颜色字符串，支持格式：RRGGBB、#RRGGBB、RRGGBBAA或#RRGGBBAA
118 /// @return 转换后的ColorRGBA对象，如果格式无效则返回白色(#FFFFFF)
119 /// @details 算法步骤：
120 /// 1. 移除可选的'#'前缀
121 /// 2. 验证长度必须是6(RGB)或8(RGBA)
122 /// 3. 将每两个字符转换为一个字节值(0-255)
123 /// 4. 检测转换错误：值为0但对应的十六进制字符不是"00"
124 static constexpr ColorRGBA hexToRGBA(std::string_view hex) noexcept {
125 // 移除'#'前缀
126 if (!hex.empty() && hex[0] == '#')
127 hex.remove_prefix(1);
128
129 const auto len = hex.size();
130 // 验证长度
131 if (len != 6 && len != 8) {
132 return ColorRGBA{255, 255, 255, 1.0f}; // 无效格式，返回白色
133 }
134
135 // 转换RGB分量
136 const i32 r = hexByteToU8(hex.substr(0, 2));
137 const i32 g = hexByteToU8(hex.substr(2, 2));
138 const i32 b = hexByteToU8(hex.substr(4, 2));
139
140 // 转换Alpha分量（如果存在）
141 const f32 a = (len == 8) ? rgbNorm[hexByteToU8(hex.substr(6, 2))] : 1.0f;
142
143 // 验证转换结果：检查是否因无效字符导致转换失败（返回0但hex不是"00"）
144 const auto isValidHexByte = [&hex](i32 value, size_t pos1, size_t pos2) {
145 return !(value == 0 && (hex[pos1] != '0' || hex[pos2] != '0'));
146 };
147
148 if (!isValidHexByte(r, 0, 1) || !isValidHexByte(g, 2, 3) || !isValidHexByte(b, 4, 5) ||
149 (len == 8 && !isValidHexByte(hexByteToU8(hex.substr(6, 2)), 6, 7))) {
150 return ColorRGBA{255, 255, 255, 1.0f}; // 无效字符，返回白色
151 }
152
153 return ColorRGBA{r, g, b, a};
154 }

166 /// @brief 将两个十六进制字符转换为对应的字节值(0-255)
167 /// @param sv 包含1-2个十六进制字符的字符串视图
168 /// @return 转换后的字节值，如果包含无效字符则返回0
169 /// @details 支持大写和小写十六进制字符(A-F, a-f)。算法从左到右处理每个字符，
170 /// 每次将当前值乘以16并加上新字符的值。
171 static constexpr i32 hexByteToU8(const std::string_view sv) noexcept {
172 i32 value = 0;
173 for (const char c: sv) {
174 value *= 16;
175 if (c >= '0' && c <= '9') {
176 value += static_cast<i32>(c - '0');
177 } else if (c >= 'A' && c <= 'F') {
178 value += static_cast<i32>(c - 'A' + 10);
179 } else if (c >= 'a' && c <= 'f') {
180 value += static_cast<i32>(c - 'a' + 10);
181 } else {
182 return 0; // 无效字符
183 }
184 }
185 return value;
186 }
