14 inline std::string currentDateTime() {
15 const auto now = std::chrono::system_clock::now();
16 const auto now_time_t = std::chrono::system_clock::to_time_t(now);
17 const auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(
18 now.time_since_epoch()) % 1000;
19 const std::tm *local_tm = std::localtime(&now_time_t);
20 return std::format("{:04}-{:02}-{:02} {:02}:{:02}:{:02}.{:03}",
21 local_tm->tm_year + 1900, local_tm->tm_mon + 1, local_tm->tm_mday,
22 local_tm->tm_hour, local_tm->tm_min, local_tm->tm_sec, ms.count());
23 }
