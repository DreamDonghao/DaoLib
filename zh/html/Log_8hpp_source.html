14 inline std::string currentDateTime() {
15 using namespace std::chrono;
16
17 const auto now = system_clock::now();
18 const zoned_time china_time{"Asia/Shanghai", now};
19 const auto sec = floor<seconds>(china_time.get_local_time());
20 const auto ms = duration_cast<milliseconds>(china_time.get_local_time() - sec).count();
21
22 return std::format("{:%Y-%m-%d %H:%M:%S}.{:03}", sec, ms);
23 }

151 static constexpr std::string_view Color(const LogLevel level) {
152 using enum LogLevel;
153 switch (level) {
154 case TRACE: return fmt_color::dim; ///< 暗淡，低优先级
155 case DEBUG: return fmt_color::bright_cyan; ///< 亮青色，调试信息
156 case INFO: return fmt_color::bright_green; ///< 亮绿色，正常状态
157 case WARN: return fmt_color::bright_yellow; ///< 亮黄色，警告
158 case ERROR: return fmt_color::bright_red; ///< 亮红色，错误
159 case FATAL: return "\033[1;41;97m"; ///< 粗体+红底+亮白字，致命错误
160 default: return fmt_color::reset;
161 }
162 }
