15 /// @brief 固定大小线程池
16 /// @details 停止协议: ~ThreadPool() 设 m_stop → notify_all → 成员逆序析构
17 /// 成员声明顺序决定析构顺序，m_threads 在最后 → 最先销毁，join 时 mutex/cv 仍存活
18 class ThreadPool {
19 public:
20 explicit ThreadPool(const size_t numThreads = 4) {
21 for (size_t i = 0; i < numThreads; ++i) {
22 m_threads.emplace_back([this](const std::stop_token &st) {
23 while (true) {
24 std::function<void()> task;
25 {
26 std::unique_lock lock(m_mutex);
27 m_cv.wait(lock, st, [this] {
28 return m_stop || !m_tasks.empty();
29 });
30 if ((m_stop && m_tasks.empty()) || st.stop_requested())
31 return;
32 task = std::move(m_tasks.front());
33 m_tasks.pop();
34 }
35 if (task) task();
36 }
37 });
38 }
39 }
40
41 ~ThreadPool() {
42 {
43 std::lock_guard lock(m_mutex);
44 m_stop = true;
45 }
46 m_cv.notify_all();
47 }
48
49 void enqueue(std::function<void()> task) {
50 {
51 std::lock_guard lock(m_mutex);
52 m_tasks.push(std::move(task));
53 }
54 m_cv.notify_one();
55 }
56
57 private:
58 std::atomic<bool> m_stop{false};
59 std::mutex m_mutex;
60 std::condition_variable_any m_cv;
61 std::queue<std::function<void()> > m_tasks;
62 std::vector<std::jthread> m_threads; // 最后声明，最先析构（join 时依赖前面所有成员存活）
63 };

99 /// @brief 同步执行 HTTP 请求，按 HttpMethod 分发到 httplib 对应方法
100 HttpResponse doRequest(const HttpRequest &req) {
101 HttpResponse resp;
102 try {
103 withClient([&](auto &c) {
104 httplib::Result res;
105 const auto hdrs = toHttplibHeaders(req.headers);
106 switch (req.method) {
107 case HttpMethod::Get:
108 res = c.Get(req.path, hdrs);
109 break;
110 case HttpMethod::Post:
111 res = c.Post(req.path, hdrs, req.body, req.content_type);
112 break;
113 case HttpMethod::Put:
114 res = c.Put(req.path, hdrs, req.body, req.content_type);
115 break;
116 case HttpMethod::Delete:
117 res = c.Delete(req.path, hdrs, req.body, req.content_type);
118 break;
119 case HttpMethod::Patch:
120 res = c.Patch(req.path, hdrs, req.body, req.content_type);
121 break;
122 }
123 if (res) {
124 resp.status = res->status;
125 resp.body = std::move(res->body);
126 resp.success = (res->status == 200);
127 if (res->status != 200)
128 resp.error = "HTTP Error: " + std::to_string(res->status);
129 } else {
130 resp.error = "Request failed: " + httplib::to_string(res.error());
131 }
132 });
133 } catch (const std::exception &e) {
134 resp.error = std::string("Exception: ") + e.what();
135 }
136 return resp;
137 }
