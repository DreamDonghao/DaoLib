29 void handleEvent(const SDL_Event &event) {
30 if (m_status == InputStatus::Disabled) return;
31
32 f32 mx = 0, my = 0;
33 if (event.type == SDL_EVENT_MOUSE_MOTION) {
34 mx = event.motion.x; my = event.motion.y;
35 } else if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN || event.type == SDL_EVENT_MOUSE_BUTTON_UP) {
36 mx = event.button.x; my = event.button.y;
37 }
38
39 SDL_Window *win = SDL_GetWindowFromID(event.window.windowID);
40
41 if (m_status == InputStatus::Input) {
42 // 更新 IME 光标位置
43 const SDL_Rect rect{
44 static_cast<int>(m_boundingBox.getLeft()),
45 static_cast<int>(m_boundingBox.getTop()),
46 static_cast<int>(m_boundingBox.getWidth()),
47 static_cast<int>(m_boundingBox.getHeight())
48 };
49 SDL_SetTextInputArea(win, &rect, m_cursorPos);
50
51 if (event.type == SDL_EVENT_TEXT_INPUT) {
52 deleteSelection();
53 const char *utf8chars = event.text.text;
54 const auto begin = utf8::unchecked::iterator<const char *>(utf8chars);
55 const auto end = utf8::unchecked::iterator<const char *>(utf8chars + std::strlen(utf8chars));
56 for (auto it = begin; it != end; ++it) {
57 m_str.insert(m_str.begin() + m_cursorPos, static_cast<char32_t>(*it));
58 ++m_cursorPos;
59 }
60 } else if (event.type == SDL_EVENT_KEY_DOWN) {
61 handleKeyDown(event.key, win);
62 } else if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN && event.button.button == SDL_BUTTON_LEFT) {
63 if (!m_boundingBox.isInBoundingBox(mx, my)) {
64 m_status = InputStatus::Normal;
65 m_selectionStart = -1;
66 SDL_StopTextInput(win);
67 }
68 }
69 } else if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN && event.button.button == SDL_BUTTON_LEFT) {
70 if (m_boundingBox.isInBoundingBox(mx, my)) {
71 m_status = InputStatus::Input;
72 m_selectionStart = -1;
73 m_cursorBlinkTime = 0;
74 m_cursorVisible = true;
75 SDL_StartTextInput(win);
76 }
77 }

89 void writeToBatch(BatchRenderer &batchRenderer) const override {
90 // 背景和边框
91 if (m_status == InputStatus::Input) {
92 m_rectBorder.setColor(ColorRGBA(100, 180, 255, 1.0f));
93 } else {
94 m_rectBorder.setColor(ColorRGBA(100, 100, 100, 1.0f));
95 }
96 m_rectBg.writeToBatch(batchRenderer);
97 m_rectBorder.writeToBatch(batchRenderer);
98
99 const f32 pad = 4;
100 const f32 visibleLeft = m_boundingBox.getLeft() + pad;
101 const f32 visibleRight = m_boundingBox.getRight() - pad;
102 const f32 visibleWidth = visibleRight - visibleLeft;
103
104 const auto &atlas = GlyphAtlas::getGlyphAtlas();
105 const f32 fontScale = m_text.lineHeight() / atlas.getFontSize();
106
107 // 计算光标在文本空间中的 x 坐标（从文本基线起算）
108 f32 cursorLocalX = 0;
109 for (i32 i = 0; i < m_cursorPos && i < static_cast<i32>(m_str.size()); ++i) {
110 cursorLocalX += atlas.getGlyphAdvance(m_str[i]) * fontScale;
111 }
112
113 // 调整滚动，保持光标可见
114 const f32 cursorScreenX = visibleLeft - m_scrollX + cursorLocalX;
115 if (cursorScreenX > visibleRight) {
116 m_scrollX = cursorLocalX - (visibleWidth - 1);
117 } else if (cursorScreenX < visibleLeft) {
118 m_scrollX = cursorLocalX;
119 }
120 m_scrollX = std::max(0.0f, m_scrollX);
121
122 m_cursorPixelX = visibleLeft - m_scrollX + cursorLocalX;
123
124 // 设置文本位置和裁剪参数
125 m_text.setPosition(visibleLeft, m_text.y());
126 m_text.setWidth(visibleWidth);
127 m_text.setScrollX(m_scrollX);
128
129 m_placeholder.setPosition(visibleLeft, m_placeholder.y());
130 m_placeholder.setWidth(visibleWidth);
131
132 // 文本或 placeholder
133 if (!m_str.empty()) {
134 m_text.setContent(m_str);
135 m_text.writeToBatch(batchRenderer);
136 } else if (m_status != InputStatus::Input) {
137 m_placeholder.setContent(m_placeholderText);
138 m_placeholder.writeToBatch(batchRenderer);
139 }
140
141 // 选区高亮
142 if (m_selectionStart >= 0 && m_selectionStart != m_cursorPos) {
143 const i32 selStart = std::min(m_selectionStart, m_cursorPos);
144 const i32 selEnd = std::max(m_selectionStart, m_cursorPos);
145
146 f32 selLocalX = 0, endLocalX = 0;
147 for (i32 i = 0; i < selEnd; ++i) {
148 if (i == selStart) selLocalX = endLocalX;
149 if (i < static_cast<i32>(m_str.size())) {
150 endLocalX += atlas.getGlyphAdvance(m_str[i]) * fontScale;
151 }
152 }
153
154 const f32 selX = visibleLeft - m_scrollX + selLocalX;
155 const f32 selW = visibleLeft - m_scrollX + endLocalX - selX;
156 if (selW > 0) {
157 const SDL_FColor selColor{0.2f, 0.4f, 0.8f, 0.5f};
158 SDL_Vertex *v = batchRenderer.allocateVertices(0, 6);
159 v[0] = {{selX, m_text.y()}, selColor, {0, 0}};
160 v[1] = {{selX + selW, m_text.y()}, selColor, {1, 0}};
161 v[2] = {{selX + selW, m_text.y() + m_text.lineHeight()}, selColor, {1, 1}};
162 v[3] = v[2];
163 v[4] = {{selX, m_text.y() + m_text.lineHeight()}, selColor, {0, 1}};
164 v[5] = v[0];
165 }
166 }
167
168 // 光标线
169 if (m_status == InputStatus::Input && m_cursorVisible) {
170 const f32 cy = m_text.y() + 2;
171 const f32 ch = m_text.lineHeight() - 4;
172 const SDL_FColor cColor{1, 1, 1, 1};
173 SDL_Vertex *v = batchRenderer.allocateVertices(0, 6);
174 const f32 cw = 1.5f;
175 v[0] = {{m_cursorPixelX, cy}, cColor, {0, 0}};
176 v[1] = {{m_cursorPixelX + cw, cy}, cColor, {1, 0}};
177 v[2] = {{m_cursorPixelX + cw, cy + ch}, cColor, {1, 1}};
178 v[3] = v[2];
179 v[4] = {{m_cursorPixelX, cy + ch}, cColor, {0, 1}};
180 v[5] = v[0];

254 void handleKeyDown(const SDL_KeyboardEvent &key, SDL_Window *win) {
255 const bool ctrl = (key.mod & SDL_KMOD_CTRL) != 0;
256 const bool shift = (key.mod & SDL_KMOD_SHIFT) != 0;
257
258 // Ctrl shortcuts
259 if (ctrl && key.key == SDLK_A) { selectAll(); return; }
260 if (ctrl && key.key == SDLK_C) { copyToClipboard(); return; }
261 if (ctrl && key.key == SDLK_X) { copyToClipboard(); deleteSelection(); return; }
262 if (ctrl && key.key == SDLK_V) { pasteFromClipboard(); return; }
263
264 if (key.key == SDLK_BACKSPACE && !m_str.empty()) {
265 if (m_selectionStart >= 0) { deleteSelection(); return; }
266 if (m_cursorPos > 0) { m_str.erase(m_cursorPos - 1, 1); --m_cursorPos; }
267 } else if (key.key == SDLK_DELETE) {
268 if (m_selectionStart >= 0) { deleteSelection(); return; }
269 if (m_cursorPos < static_cast<i32>(m_str.size())) { m_str.erase(m_cursorPos, 1); }
270 } else if (key.key == SDLK_LEFT) {
271 if (m_selectionStart < 0 && shift) m_selectionStart = m_cursorPos;
272 m_cursorPos = std::max(0, m_cursorPos - 1);
273 if (!shift) m_selectionStart = -1;
274 } else if (key.key == SDLK_RIGHT) {
275 if (m_selectionStart < 0 && shift) m_selectionStart = m_cursorPos;
276 m_cursorPos = std::min(static_cast<i32>(m_str.size()), m_cursorPos + 1);
277 if (!shift) m_selectionStart = -1;
278 } else if (key.key == SDLK_HOME) {
279 if (m_selectionStart < 0 && shift) m_selectionStart = m_cursorPos;
280 m_cursorPos = 0;
281 if (!shift) m_selectionStart = -1;
282 } else if (key.key == SDLK_END) {
283 if (m_selectionStart < 0 && shift) m_selectionStart = m_cursorPos;
284 m_cursorPos = static_cast<i32>(m_str.size());
285 if (!shift) m_selectionStart = -1;
286 } else if (key.key == SDLK_RETURN || key.key == SDLK_ESCAPE) {
287 m_status = InputStatus::Normal;
288 m_selectionStart = -1;
289 SDL_StopTextInput(win);
290 }
291 }
